# Пример структуры Secret (значения — заглушки). Настоящие Secret'ы создаёт scripts/create-secrets.ps1
# из файлов lab2/secrets; этот файл kubectl не применяет (расширение .txt).
apiVersion: v1
kind: Secret
metadata:
  name: booking-service-secrets
  namespace: fitclub
type: Opaque
stringData:
  db_booking_password: "<пароль роли booking_user>"
  rabbitmq_booking_password: "<пароль пользователя RabbitMQ booking_service>"
  service_token_booking_to_membership: "<служебный токен booking → membership>"
  jwt_public_key.pem: |
    -----BEGIN PUBLIC KEY-----
    <публичный ключ RS256 из lab2/secrets/jwt_public_key.pem>
    -----END PUBLIC KEY-----
